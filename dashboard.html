展会客户汇总
— 位客户
⚙️ 一次性配置（负责人操作，约5分钟）
配置完成后，员工提交的每条客户信息会实时出现在下方列表，无需手动传文件。
打开 script.google.com，点击「新建项目」
把下方的代码完整粘贴进去，替换掉默认的 myFunction，然后点保存
点击「部署」→「新建部署」→ 类型选「Web 应用」→ 执行身份选「我」→ 访问权限选「所有人」→ 点「部署」→ 复制生成的网址
把网址粘贴到下方输入框，点「保存并开始使用」。同时把这个网址发给所有员工填入他们的录入工具。
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.openById('YOUR_SHEET_ID');
    var sheet = ss.getSheetByName('客户数据') || ss.insertSheet('客户数据');

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['ID','提交时间','公司名称','联系人','职位','市场','采购能力/规模',
        '邮件','WhatsApp/电话','社媒账号','选款/大底分类','打样要求',
        '预计下单量','决策周期','客户痛点','样品图数量','合照数量','跟进状态','备注',
        '样品图(base64)','合照(base64)']);
    }

    var styleImgs = (data.styleImages || []).join('|||');
    var portraitImgs = (data.portraitImages || []).join('|||');

    sheet.appendRow([
      data.id, data.exportedAt, data.company, data.name, data.jobTitle,
      data.market, data.scale, data.email, data.phone, data.social,
      data.cats, data.sample, data.qty, data.timeline, data.pain,
      (data.styleImages || []).length, (data.portraitImages || []).length,
      data.status || '新客户', data.note, styleImgs, portraitImgs
    ]);

    return ContentService.createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.openById('YOUR_SHEET_ID');
  var sheet = ss.getSheetByName('客户数据');
  if (!sheet || sheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    headers.forEach(function(h, j) { obj[h] = rows[i][j]; });
    result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
⚠️ 记得把代码里两处 YOUR_SHEET_ID 替换成你的 Google Sheets 文件 ID（URL 中 /d/ 和 /edit 之间的部分）
